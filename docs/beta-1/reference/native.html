/// Encode an identifier as a single URL path segment, rejecting values that
/// could escape the intended path.
pub(crate) fn encode_segment(id: &str) -> Result<String, DomainError> {
    if id.is_empty()
        || id == "."
        || id == ".."
        || id.contains('/')
        || id.contains('\\')
        || id.chars().any(|c| c.is_control())
    {
        return Err(DomainError::denied());
    }

    let mut out = String::with_capacity(id.len());
    for b in id.bytes() {
        match b {
            b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'-' | b'.' | b'_' | b'~' => {
                out.push(b as char)
            }
            _ => out.push_str(&format!("%{:02X}", b)),
        }
    }
    Ok(out)
}

pub async fn execute(request: CodeRequest, proof: String) -> Result<Value, DomainError> {
    let base = Url::parse(BASE).map_err(|_| DomainError::denied())?;

    let (method, segment, body, params): (Method, String, Option<Value>, Vec<(&str, String)>) =
        match request {
            CodeRequest::Connections {} => {
                (Method::GET, "connections".to_string(), None, Vec::new())
            }
            CodeRequest::Connection { id } => (
                Method::GET,
                format!("connections/{}", encode_segment(&id)?),
                None,
                Vec::new(),
            ),
            CodeRequest::SendProposal { payload } => (
                Method::POST,
                "connections".to_string(),
                Some(serde_json::json!(payload)),
                Vec::new(),
            ),
            CodeRequest::CloseConnection { id, payload } => (
                Method::POST,
                format!("connections/{}/closeout", encode_segment(&id)?),
                Some(serde_json::json!(payload)),
                Vec::new(),
            ),
            CodeRequest::Workspace {} => (Method::GET, "workspace".to_string(), None, Vec::new()),
            CodeRequest::Sources {} => (Method::GET, "sources".to_string(), None, Vec::new()),
            CodeRequest::Import { payload } => (
                Method::POST,
                "imports".to_string(),
                Some(payload),
                Vec::new(),
            ),
            CodeRequest::Lookup {
                section_number,
                as_of,
            } => {
                let mut q = vec![("section_number", section_number)];
                if let Some(a) = as_of {
                    q.push(("as_of", a));
                }
                (Method::GET, "lookup".to_string(), None, q)
            }
            CodeRequest::Citation {
                section_number,
                as_of,
            } => {
                let mut q = vec![("section_number", section_number)];
                if let Some(a) = as_of {
                    q.push(("as_of", a));
                }
                (Method::GET, "citation".to_string(), None, q)
            }
            CodeRequest::Search { query } => {
                (Method::GET, "search".to_string(), None, vec![("q", query)])
            }
            CodeRequest::Export { section_ref, as_of } => {
                let mut params: Vec<(&str, String)> = vec![("section_ref", section_ref.clone())];
                if let Some(v) = as_of.clone() {
                    params.push(("as_of", v));
                }
                (reqwest::Method::GET, "export".to_string(), None, params)
            }
            CodeRequest::Question {
                question,
                section_number,
                as_of,
            } => {
                let body = serde_json::json!({
                    "question": question,
                    "section_number": section_number,
                    "as_of": as_of,
                });
                (
                    reqwest::Method::POST,
                    "questions".to_string(),
                    Some(body),
                    Vec::new(),
                )
            }
            CodeRequest::ImportTree { id } => (
                reqwest::Method::GET,
                format!("imports/{}/tree", encode_segment(&id)?),
                None,
                Vec::new(),
            ),
            CodeRequest::ImportProvenance { id } => (
                reqwest::Method::GET,
                format!("imports/{}/provenance", encode_segment(&id)?),
                None,
                Vec::new(),
            ),
            CodeRequest::History { id } => (
                Method::GET,
                format!("sections/{}/history", encode_segment(&id)?),
                None,
                Vec::new(),
            ),
        };

    let mut url = base
        .join(&format!("/desktop/v1/code/{}", segment))
        .map_err(|_| DomainError::denied())?;

    if !params.is_empty() {
        let mut pairs = url.query_pairs_mut();
        for (k, v) in &params {
            pairs.append_pair(k, v);
        }
    }

    let path_and_query = match url.query() {
        Some(q) => format!("{}?{}", url.path(), q),
        None => url.path().to_string(),
    };

    send(method, path_and_query, body, &proof).await
}

pub async fn execute_access(request: AccessRequest, proof: String) -> Result<Value, DomainError> {
    match request {
        AccessRequest::ExportReview { review_id } => {
            let plan = send(
                Method::POST,
                format!("/desktop/v1/access/reviews/{review_id}/export"),
                None,
                &proof,
            )
            .await?;
            let plan: crate::domain_files::ArtifactPlan =
                serde_json::from_value(plan).map_err(|_| DomainError::transport(true))?;
            let artifact_id = plan.artifact_id;
            let path = tauri::async_runtime::spawn_blocking(move || {
                crate::domain_files::write_access_export(plan)
            })
            .await
            .map_err(|_| DomainError::transport(true))?
            .map_err(|_| {
                DomainError::new(
                    "export-incomplete",
                    "Access export was not finalized. Prepared evidence is retained.",
                    true,
                )
            })?;
            send(
                Method::POST,
                format!("/desktop/v1/access/exports/{artifact_id}/complete"),
                Some(serde_json::json!({"export_path":path})),
                &proof,
            )
            .await
        }
        AccessRequest::Tool { payload } => {
            send(
                Method::POST,
                "/desktop/v1/access/tools".into(),
                Some(serde_json::json!(payload)),
                &proof,
            )
            .await
        }
        AccessRequest::Rewrite { payload } => {
            let context = send(
                Method::POST,
                "/desktop/v1/access/rewrite-context".into(),
                Some(serde_json::json!(&payload)),
                &proof,
            )
            .await?;
            let prompt = context
                .get("prompt")
                .and_then(Value::as_str)
                .ok_or_else(|| DomainError::transport(false))?
                .to_owned();
            let generated = tauri::async_runtime::spawn_blocking(move || {
                if crate::model::local_generation_available()? {
                    crate::model::generate_local_text(&prompt).map(Some)
                } else { Ok(None) }
            }).await.map_err(|_| DomainError::transport(false))?
                .map_err(|_| DomainError::new("model-unavailable", "Local AI generation failed. No rewrite was recorded; no sample was substituted.", false))?;
            let mut body = serde_json::json!(payload);
            if let Some((model, text)) = generated {
                body["outcome"] = "generated".into();
                body["generated"] = text.into();
                body["model"] = model.into();
            } else {
                body["outcome"] = "unavailable".into();

        }
        AccessRequest::State {} => {
            send(Method::GET, "/desktop/v1/access/state".into(), None, &proof).await
        }
        AccessRequest::DeleteReview { review_id } => {
            send(
                Method::POST,
                format!("/desktop/v1/access/reviews/{review_id}/delete"),
                None,
                &proof,
            )
            .await
        }
        AccessRequest::Review { payload } => {
            let context = send(
                Method::POST,
                "/desktop/v1/access/review-context".into(),
                Some(serde_json::json!(&payload)),
                &proof,
            )
            .await?;
            let prompt = context
                .get("prompt")
                .and_then(Value::as_str)
                .ok_or_else(|| DomainError::transport(false))?
                .to_owned();
            // Runtime availability and generation only; findings/prompt and
            // persistence belong to Access. Failed AI must not lose a review.
            let generated = tauri::async_runtime::spawn_blocking(move || {
                if crate::model::local_generation_available().unwrap_or(false) {
                    Some(crate::model::generate_local_text(&prompt))
                } else {
                    None
                }
            })
            .await;
            let mut body = serde_json::json!(&payload);
            match generated {
                Ok(Some(Ok((model, text)))) => {
                    body["ai_outcome"] = "generated".into();
                    body["ai_analysis"] = text.into();
                    body["ai_analysis_model"] = model.into();
                }
                Ok(None) => body["ai_outcome"] = "unavailable".into(),
                _ => body["ai_outcome"] = "failed".into(),
            }
            send(
                Method::POST,
                "/desktop/v1/access/reviews".into(),
                Some(body),
                &proof,
            )
            .await
        }
    }

fn status_error(status: StatusCode, writing: bool) -> DomainError {
    match status.as_u16() {
        401 | 403 => DomainError::denied(),
        400 | 422 => DomainError::new("invalid-input", "Records rejected the input. Review the required fields; no change was accepted.", false),
        404 => DomainError::new("not-found", "The Records request or operation was not found. Refresh before continuing.", false),
        409 => DomainError::new("conflict", "Records reported a conflict. Refresh before continuing.", false),
        503 => DomainError::new("not-ready", "Records is not ready or migration is not activated. Refresh after service readiness is restored.", writing),
        _ => DomainError::transport(writing),
    }
}

pub async fn execute(request: RecordsRequest, proof: String) -> Result<Value, DomainError> {
    let request = match request {
        RecordsRequest::Suggest { request_id } => {
            #[derive(Deserialize)]
            #[serde(deny_unknown_fields)]
            struct DraftContext {
                prompt: String,
                previous_response_sha256: String,
            }
            let context = send(
                Method::GET,
                format!("/desktop/v1/requests/{request_id}/draft-context"),
                None,
                &proof,
            )
            .await?;
            let context: DraftContext =
                serde_json::from_value(context).map_err(|_| DomainError::transport(false))?;
            let generated = tauri::async_runtime::spawn_blocking(move || crate::model::generate_local_text(&context.prompt))
                .await.map_err(|_| DomainError::transport(false))?
                .map_err(|_| DomainError::new("model-unavailable", "Local AI could not generate a draft. The existing text and approval were not changed.", false))?;
            return send(Method::POST, format!("/desktop/v1/requests/{request_id}/generated-draft"),
                        Some(serde_json::json!({"text":generated.1, "model":generated.0, "previous_response_sha256":context.previous_response_sha256})), &proof).await;
        }
        RecordsRequest::Export { request_id, kind } => {
            let plan = send(
                Method::POST,
                format!("/desktop/v1/requests/{request_id}/artifacts"),
                Some(serde_json::json!({"kind":kind})),
                &proof,
            )
            .await?;
            let plan: crate::domain_files::ArtifactPlan =
                serde_json::from_value(plan).map_err(|_| DomainError::transport(true))?;
            let artifact_id = plan.artifact_id;
            let path = tauri::async_runtime::spawn_blocking(move || crate::domain_files::write_export(plan))
                .await.map_err(|_| DomainError::transport(true))?
                .map_err(|_| DomainError::new("export-incomplete", "The Records export was not finalized. Prepared evidence is retained; refresh before retrying.", true))?;
            return send(
                Method::POST,
                format!("/desktop/v1/requests/{request_id}/artifacts/{artifact_id}/complete"),
                Some(serde_json::json!({"export_path":path})),
                &proof,
            )
            .await;
        }
        other => other,
    };
    let (method, path, body) = tauri::async_runtime::spawn_blocking(move || request.wire())
        .await
        .map_err(|_| DomainError::transport(false))??;
    send(method, path, body, &proof).await
}

pub(crate) async fn send(
    method: Method,
    path: String,
    body: Option<Value>,
    proof: &str,
) -> Result<Value, DomainError> {
    send_with_header(method, path, body, proof, "X-Townlight-Session").await
}

async fn send_with_header(
    method: Method,
    path: String,
    body: Option<Value>,
    proof: &str,
    proof_header: &'static str,
) -> Result<Value, DomainError> {
    let writing = method == Method::POST;
    // Do not forward the proof through a system proxy or a server redirect.
    let client = Client::builder()
        .no_proxy()
        .redirect(reqwest::redirect::Policy::none())
        .retry(reqwest::retry::never())
        .connect_timeout(Duration::from_secs(3))
        .timeout(Duration::from_secs(60))
        .build()
        .map_err(|_| DomainError::transport(false))?;
    let mut outgoing = client
        .request(method, format!("{DOMAIN_ORIGIN}{path}"))
        .header(proof_header, proof);
    if let Some(body) = body {
        outgoing = outgoing.json(&body);
    }
    let mut response = outgoing
        .send()
        .await
        .map_err(|_| DomainError::transport(writing))?;
    if !response.status().is_success() {
        return Err(status_error(response.status(), writing));
    }
    // Error bodies are deliberately not echoed: framework errors can contain
    // internal details. Successful responses must be bounded JSON.
    let mut bytes = Vec::new();
    while let Some(chunk) = response
        .chunk()
        .await
        .map_err(|_| DomainError::transport(writing))?
    {
        if bytes.len().saturating_add(chunk.len()) > MAX_RESPONSE_BYTES {
            return Err(DomainError::new("response-too-large", "Records returned more data than this client can safely load. No partial state was applied.", writing));
        }
        bytes.extend_from_slice(&chunk);
    }
    serde_json::from_slice(&bytes).map_err(|_| {
        DomainError::new(
            "invalid-response",
            "Records returned an unreadable result. Refresh before repeating a change.",
            writing,
        )
    })
}

pub async fn execute(request: InspectionsRequest, proof: String) -> Result<Value, DomainError> {
    let export = match &request {
        InspectionsRequest::ExportPacket {
            id,
            expected_sha256,
        } => Some((*id, expected_sha256.clone())),
        _ => None,
    };
    let (method, path, body) = route(request);
    let result = send(method, path, body, &proof).await?;
    if let Some((id, expected)) = export {
        let content = result
            .get("text")
            .and_then(Value::as_str)
            .ok_or_else(DomainError::denied)?;
        let digest = result
            .get("sha256")
            .and_then(Value::as_str)
            .ok_or_else(DomainError::denied)?;
        if digest != expected {
            return Err(DomainError::denied());
        }
        let path = crate::domain_files::write_inspections_file(
            &format!("inspections-{id}-{}.json", Uuid::new_v4()),
            content.as_bytes(),
            digest,
        )
        .map_err(|_| DomainError::denied())?;
        return Ok(serde_json::json!({"path": path, "sha256": digest}));
    }
    Ok(result)
}

fn require_enabled_city_modules(action: &str, payload: Option<&Value>) -> Result<(), String> {
    let Some((module_ids, allow_any)) = city_work_action_module_requirement(action, payload) else {
        return Ok(());
    };
    let selection = module_registry::module_selection_state()?;
    let enabled = module_ids
        .iter()
        .filter(|module_id| {
            selection
                .enabled_module_ids
                .iter()
                .any(|enabled_id| enabled_id == **module_id)
        })
        .copied()
        .collect::<Vec<_>>();
    if (allow_any && !enabled.is_empty()) || (!allow_any && enabled.len() == module_ids.len()) {
        return Ok(());
    }
    let missing = module_ids
        .iter()
        .filter(|module_id| !enabled.iter().any(|enabled_id| enabled_id == *module_id))
        .map(|module_id| module_display_name(module_id))
        .collect::<Vec<_>>();
    Err(format!(
        "{} {} not enabled in this local profile. Install or enable {} in Settings before using this workflow.",
        missing.join(", "),
        if missing.len() == 1 { "is" } else { "are" },
        if missing.len() == 1 { "it" } else { "them" }
    ))
}

fn app_state() -> Result<AppState, String> {
    let access = auth::access_state()?;
    let admin_signed_in = access_is_local_admin(&access);
    let signed_in = access_is_signed_in(&access);
    let authority = records_authority::read()?;
    let domain_authority = match authority.as_ref().map(|journal| journal.phase.as_str()) {
        Some("active") => "python-records",
        Some(_) => "records-migration-pending",
        None => "legacy",
    };
    let city_work = if authority.is_some() {
        // The typed async client loads canonical state; never read old JSON here.
        CityWorkState::default()
    } else if signed_in {
        workflows::city_work_state()?
    } else {
        workflows::public_city_work_state()?
    };
    let users = if admin_signed_in {
        auth::saved_users()?
    } else {
        Vec::new()
    };
    let model = if admin_signed_in {
        model::model_state()?
    } else {
        public_model_state(model::model_state()?)
    };
    let health = if admin_signed_in {
        supervisor::runtime_health()?
    } else {
        public_runtime_health(supervisor::runtime_health()?)
    };
    Ok(AppState {
        domain_authority,
        product_name: "Townlight",
        status_label: "Windows Local 1.0 desktop",
        local_only: true,
        navigation: navigation(),
        modules: module_summaries()?,
        module_profiles: module_registry::module_profiles()?,
        module_selection: module_registry::module_selection_state()?,
        installer_steps: installer_steps(),
        first_run: first_run::first_run_state(&[])?,
        city_profile: first_run::saved_city_profile()?,
        users,
        access,
        model,
        health,
        city_work,
    })
}

#[tauri::command]
fn module_action(action: String, module_id: String) -> Result<ModuleActionResult, String> {
    let access = auth::access_state()?;
    if access.configured && !access_is_local_admin(&access) {
        return Err(
            "Sign in as the Townlight admin before changing installed modules.".to_string(),
        );
    }
    let previous_selection = module_registry::module_selection_state()?;
    let was_installed = previous_selection
        .installed_module_ids
        .iter()
        .any(|installed_id| installed_id == &module_id);
    let display_name = module_display_name(&module_id);
    let mut removal_backup_message = None;
    let selection = match action.as_str() {
        "enable-module" => module_registry::set_module_enabled(&module_id, true)?,
        "disable-module" => module_registry::set_module_enabled(&module_id, false)?,
        "install-module" => module_registry::set_module_installed(&module_id, true)?,
        "remove-module" => {
            if was_installed {
                let backup = supervisor::supervisor_action("backup", None)?;
                removal_backup_message = Some(backup.message);
                module_registry::set_module_installed(&module_id, false)?
            } else {
                previous_selection
            }
        }
        "update-module" => previous_selection,
        "open-module-exports" => {
            if !was_installed {
                return Err(format!(
                    "{display_name} is not installed in this local profile. Install it before opening its exports folder."
                ));
            }
            let Some((folder, _label)) = module_exports_folder(&module_id) else {
                return Err(format!(
                    "{display_name} does not have a module exports folder."
                ));
            };
            let path = local_paths::data_root().join("exports").join(folder);
            local_shell::open_local_folder(&path)?;
            previous_selection
        }
        _ => return Err(format!("Unsupported module action: {action}")),
    };
    let modules = module_summaries()?;
    let is_installed = selection
        .installed_module_ids
        .iter()
        .any(|installed_id| installed_id == &module_id);
    let is_enabled = selection
        .enabled_module_ids
        .iter()
        .any(|enabled_id| enabled_id == &module_id);
    let (status, message, next_action) = match action.as_str() {
        "enable-module" => (
            "Module enabled",
            format!("{display_name} is enabled in the local Townlight shell."),
            "Review the module list or continue city work.",
        ),
        "disable-module" => (
            "Module disabled",
            format!("{display_name} is disabled in the local Townlight shell. Its data remains installed and can be re-enabled."),
            "Review the module list or continue city work.",
        ),
        "install-module" if was_installed => (
            "Module already installed",
            format!("{display_name} is already installed in this local profile."),
            "Enable the module if it is disabled, or continue city work.",
        ),
        "install-module" => (
            "Module installed",
            format!("{display_name} is installed and enabled in this local profile."),
            "Review the module list or open the module work area.",
        ),
        "remove-module" if !was_installed => (
            "Module not installed",
            format!("{display_name} is not installed in this local profile."),
            "Choose an available module to install.",
        ),
        "remove-module" => (
            "Module removed",
            format!(
                "{display_name} is removed from the active local profile after a verified profile backup. Existing module data was not deleted and remains covered by profile backup/restore. {}",
                removal_backup_message
                    .as_deref()
                    .unwrap_or("The profile backup completed before removal.")
            ),
            "Install the module again if staff need this work area.",
        ),
        "update-module" => (
            "Module current",
            if is_installed {
                format!("{display_name} is already on the pinned version for this module manifest.")
            } else {
                format!("{display_name} is not installed; install it before checking updates.")
            },
            "Review module versions in Settings.",
        ),
        "open-module-exports" => {
            let (folder, label) =
                module_exports_folder(&module_id).expect("module export folder already checked");
            let path = local_paths::data_root().join("exports").join(folder);
            (
                "Module exports opened",
                format!(
                    "Opened the local {label} folder for {display_name}: {}",
                    path.display()
                ),
                "Use generated export files for clerk review, public records response, or local backup evidence.",
            )
        }
        _ => unreachable!("unsupported module action already returned"),
    };
    Ok(ModuleActionResult {
        accepted: true,
        action,
        status,
        message,
        next_action: if is_enabled || !matches!(status, "Module installed") {
            next_action
        } else {
            "Enable required dependencies before opening this module."
        },
        selection,
        modules,
    })
}

fn city_work_action_authorized(
    action: String,
    payload: Option<Value>,
) -> Result<CityWorkActionResult, String> {
    let mut payload = payload;
    let access = auth::access_state()?;
    let signed_in = access_is_signed_in(&access);
    let public_action = workflows::city_work_action_allows_public(&action);
    if action == "load-demo-town" && !access_is_local_admin(&access) {
        return Err(
            "Sign in with a Townlight admin account before loading demonstration data.".to_string(),
        );
    }
    if !signed_in {
        if !access.configured {
            return Err(
                "Finish first-run setup before public city workflows are available.".to_string(),
            );
        }
        if !public_action {
            auth::require_signed_in_session()?;
        }
        if action == "answer-code-question" {
            let mut public_payload = payload.unwrap_or_else(|| serde_json::json!({}));
            let Some(fields) = public_payload.as_object_mut() else {
                return Err("Public code questions must use a JSON object payload.".to_string());
            };
            fields.insert("publicOnly".to_string(), Value::Bool(true));
            payload = Some(public_payload);
        }
    } else {
        require_role_for_city_work(&access, &action, payload.as_ref())?;
    }
    require_enabled_city_modules(&action, payload.as_ref())?;
    if module_registry::module_selection_state()?.profile_id == "meetings-beta" {
        // This development profile has a single Python Meetings authority.
        // Unmigrated shared actions stay unavailable, never write old JSON.
        return Err("Use the typed Python product workflow in this Meetings candidate; legacy actions are unavailable.".into());
    }
    let mut result = workflows::city_work_action(&action, payload.as_ref())?;
    if action == "search-city-knowledge" {
        filter_search_results_for_enabled_modules(&mut result)?;
        filter_search_results_for_access(&mut result, &access);
        result.message = format!(
            "Local search completed across enabled modules with {} result(s).",
            result.search_results.len()
        );
    }
    if !signed_in
        && public_action
        && action != "lookup-public-records-request"
        && action != "add-public-records-message"
    {
        result.state = workflows::city_work_public_projection(&result.state);
    }
    Ok(result)
}

#[tauri::command]
async fn city_work_action(
    action: String,
    payload: Option<Value>,
) -> Result<CityWorkActionResult, String> {
    tauri::async_runtime::spawn_blocking(move || {
        catch_unwind(AssertUnwindSafe(|| {
            city_work_action_authorized(action, payload)
        }))
        .map_err(|_| {
            "City workflow action stopped unexpectedly. Restart Townlight and retry; if this repeats, use Repair from System Health."
                .to_string()
        })?
    })
    .await
    .map_err(|error| format!("City workflow action could not complete: {error}"))?
}

            if let Some(window) = app.get_webview_window("main") {

                let _ = window.show();

        }))
        .on_window_event(|_window, event| {
            if matches!(event, tauri::WindowEvent::CloseRequested { .. }) {
                let _ = supervisor::supervisor_action("stop", None);
            }
        })
        .invoke_handler(tauri::generate_handler![
            get_app_state,

//! Fixed native transport. No permit rules or caller-supplied URLs/paths.
use crate::domain_client::{send, DomainError};
use reqwest::Method;
use serde::Deserialize;
use serde_json::Value;
use uuid::Uuid;

#[derive(Debug, Deserialize)]
#[serde(tag = "operation", rename_all = "kebab-case", deny_unknown_fields)]
pub enum PermittingRequest {
    Requirements { offset: u32 },
    Requirement { id: Uuid },
    ImportRequirements { payload: Value },
    Cases { offset: u32 },
    Case { id: Uuid },
    CreateCase { payload: Value },
    ReviseCase { id: Uuid, payload: Value },
    History { id: Uuid },
    CreatePacket { id: Uuid, payload: Value },
    ExportPacket { id: Uuid, expected_sha256: String },
    SearchConnections { payload: Value },
    Pathway { payload: Value },
    AttachRecords { request_id: Uuid, payload: Value },
}

pub fn route(request: PermittingRequest) -> (Method, String, Option<Value>) {
    use PermittingRequest::*;
    let (method, path, body) = match request {
        Requirements { offset } => (Method::GET, format!("/requirements?offset={offset}"), None),
        Requirement { id } => (Method::GET, format!("/requirements/{id}"), None),
        ImportRequirements { payload } => (Method::POST, "/requirements".into(), Some(payload)),
        Cases { offset } => (Method::GET, format!("/cases?offset={offset}"), None),
        Case { id } => (Method::GET, format!("/cases/{id}"), None),
        CreateCase { payload } => (Method::POST, "/cases".into(), Some(payload)),
        ReviseCase { id, payload } => (
            Method::POST,
            format!("/cases/{id}/revisions"),
            Some(payload),
        ),

pub async fn execute(
    request: PlanningRequest,
    proof: String,
    public: bool,
) -> Result<Value, DomainError> {
    let export = match &request {
        PlanningRequest::ExportSnapshot {
            id,
            expected_sha256,
        } => Some((*id, expected_sha256.clone())),
        _ => None,
    };
    let original = matches!(&request, PlanningRequest::Original { .. });
    let (method, path, body) = route(request, public)?;
    let result = send(method, path, body, &proof).await?;
    if let Some((id, expected)) = export {
        let content = result
            .get("text")
            .and_then(Value::as_str)
            .ok_or_else(DomainError::denied)?;
        let digest = result
            .get("sha256")
            .and_then(Value::as_str)
            .ok_or_else(DomainError::denied)?;
        if digest != expected {
            return Err(DomainError::denied());
        }
        let path = crate::domain_files::write_planning_file(
            &format!("planning-{id}-{}.json", Uuid::new_v4()),
            content.as_bytes(),
            digest,
        )
        .map_err(|_| DomainError::denied())?;
        return Ok(serde_json::json!({"path": path, "sha256": digest}));
    }
    if original {
        use base64::Engine;
        let encoded = result
            .get("content_base64")
            .and_then(Value::as_str)
            .ok_or_else(DomainError::denied)?;
        let bytes = base64::engine::general_purpose::STANDARD
            .decode(encoded)
            .map_err(|_| DomainError::denied())?;
        let filename = result
            .get("filename")
            .and_then(Value::as_str)
            .ok_or_else(DomainError::denied)?;
        let digest = result
            .get("sha256")
            .and_then(Value::as_str)
            .ok_or_else(DomainError::denied)?;
        let path = crate::domain_files::write_planning_file(
            &format!("{}-{filename}", Uuid::new_v4()),
            &bytes,
            digest,
        )
        .map_err(|_| DomainError::denied())?;
        return Ok(serde_json::json!({"path": path, "sha256": digest}));
    }
    Ok(result)
}
